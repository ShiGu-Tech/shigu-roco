所有主题使用动态变量系统，从 utils/theme.js 中读取：

// WXML 模板中 <view style="background: {{theme.colors.card}}; color: {{theme.colors.textPrimary}};"> 内容 </view>
// JS 中切换主题 const themeManager = require('../../utils/theme.js'); themeManager.setTheme('fairytale'); themeManager.applyTheme(this);
